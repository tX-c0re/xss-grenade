# XSS Grenade — runtime dependencies
# Python 3.8+ required (uses the walrus operator and f-strings).
#
# Install everything:   pip install -r requirements.txt
# Then install browsers for headless verification (optional but recommended):
#                       python -m playwright install chromium

# ── Core (required) ───────────────────────────────────────────────────────────
requests>=2.28.0            # HTTP client used by the crawler and all scanners
beautifulsoup4>=4.11.0      # HTML parsing / link + form extraction (the `bs4` import)
alive-progress>=3.0.0       # progress bars used by the scan engine

# ── Static JS analysis (strongly recommended) ────────────────────────────────
# Without esprima the AST-based phases (DOM XSS taint, template injection,
# prototype-pollution gadgets, postMessage origin checks, library/CVE audit)
# degrade gracefully but lose a lot of detection power.
esprima>=4.0.1              # JavaScript AST parser

# ── Accurate HTML reflection-context classification (strongly recommended) ────
# Without tree-sitter, _html_analyzer.analyze_html() falls back to a REGEX
# classifier that mis-reads reflections inside URL attributes — e.g. a query
# param reflected into href="?...&p=HERE" gets classified as an *unquoted*
# attribute (wrong severity + wrong breakout char), so real reflected XSS in
# href/src attributes is missed. The tree-sitter parser classifies these
# correctly. Install both the runtime and the HTML grammar.
tree-sitter>=0.21.0        # incremental parser runtime
tree-sitter-html>=0.20.0   # HTML grammar for the reflection-context engine

# ── Headless verification (optional, recommended) ─────────────────────────────
# Used to confirm findings in a real browser (Chromium) and cut false positives.
# After installing, run:  python -m playwright install chromium
playwright>=1.40.0

# ── GUI (required) ────────────────────────────────────────────────────────────
# XSS Grenade runs as a graphical desktop application (xss_grenade_gui.py).
PyQt5>=5.15.0

# ── Advanced TLS / fingerprint evasion (optional) ─────────────────────────────
# Enables --cffi-impersonate / --legacy-tls (browser-like TLS fingerprints to
# bypass some WAFs). The tool runs fine without it; the feature is just disabled.
curl_cffi>=0.6.0
